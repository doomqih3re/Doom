function fibonacciLRU(n, cache = new Map()) {
  if (n <= 1) return n;
  if (cache.has(n)) return cache.get(n);
  const result = fibonacciLRU(n - 1, cache) + fibonacciLRU(n - 2, cache);
  cache.set(n, result);
  return result;
}
